cmake_minimum_required(VERSION 3.21)
project(STM32LiveMonitor LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build on Windows for x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(IMGUI ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
set(IMPLOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/implot)
add_library(gui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp ${IMPLOT}/implot.cpp ${IMPLOT}/implot_items.cpp)
target_include_directories(gui PUBLIC ${IMGUI} ${IMGUI}/backends ${IMPLOT})
target_compile_definitions(gui PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(gui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
add_executable(STM32LiveMonitor src/main.cpp src/Dashboard.cpp src/Monitor.cpp src/JLink.cpp)
target_link_libraries(STM32LiveMonitor PRIVATE gui)
if(MSVC)
  target_compile_options(STM32LiveMonitor PRIVATE /W4 /utf-8 /permissive-)
endif()
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT STM32LiveMonitor)
include(CTest)
if(BUILD_TESTING)
  add_executable(MonitorTests tests/MonitorTests.cpp src/Monitor.cpp src/JLink.cpp)
  target_include_directories(MonitorTests PRIVATE src)
  target_compile_definitions(MonitorTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_test(NAME MonitorTests COMMAND MonitorTests)
endif()
